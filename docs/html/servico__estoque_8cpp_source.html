46ResVenda ServicoEstoque::registrarVenda(const std::string& codigo,
47 int quantidade) {
48 if (!sessao_ || !banco_)
49 return ResVenda::falha(Erro::AcessoNegado, "Sem sessao");
50 const Usuario& eu = sessao_->usuario();
51 if (eu.perfil() != Perfil::OperadorCaixa &&
52 eu.perfil() != Perfil::GerenteUnidade)
53 return ResVenda::falha(Erro::AcessoNegado,
54 "Perfil sem permissao para registrar venda");
55 if (quantidade <= 0)
56 return ResVenda::falha(Erro::QuantidadeInvalida,
57 "Quantidade deve ser maior que zero");
58 const Produto* p = banco_->buscarProduto(codigo);
59 if (!p)
60 return ResVenda::falha(Erro::NaoEncontrado, "Produto inexistente");
61 if (!p->temPreco())
62 return ResVenda::falha(Erro::ProdutoSemPreco, "Produto sem preco");
63
64 Data hoje = Relogio::hoje();
65 if (quantidade > banco_->saldoValido(eu.idUnidade(), codigo, hoje))
66 return ResVenda::falha(Erro::EstoqueInsuficiente,
67 "Estoque valido insuficiente");
68
69 // lotes validos da unidade, do que vence primeiro para o que vence depois
70 std::vector<LoteEstoque*> candidatos;
71 std::vector<LoteEstoque>& todos = banco_->lotes();
72 for (size_t i = 0; i < todos.size(); ++i) {
73 LoteEstoque& l = todos[i];
74 if (l.idUnidade() == eu.idUnidade() && l.codigoProduto() == codigo &&
75 l.saldo() > 0 && !l.vencido(hoje))
76 candidatos.push_back(&l);
77 }
78 std::stable_sort(candidatos.begin(), candidatos.end(), menorValidade);
79
80 int falta = quantidade;
81 for (size_t i = 0; i < candidatos.size() && falta > 0; ++i) {
82 int tira = std::min(falta, candidatos[i]->saldo());
83 candidatos[i]->baixar(tira);
84 falta -= tira;
85 }
86
87 Venda v = banco_->adicionarVenda(eu.idUnidade(), codigo, quantidade,
88 p->preco() * quantidade, hoje, eu.id());
89 return ResVenda::sucesso(v);
90}

93ServicoEstoque::consultarEstoque(const FiltroEstoque& filtro) {
94 typedef Resultado<std::vector<ItemEstoque> > ResItens;
95 if (!sessao_ || !banco_)
96 return ResItens::falha(Erro::AcessoNegado, "Sem sessao");
97
98 const Usuario& eu = sessao_->usuario();
99 Perfil perfil = eu.perfil();
100
101 int idUnidade = 0;
102 if (perfil == Perfil::RepositorEstoque || perfil == Perfil::GerenteUnidade) {
103 if (filtro.idUnidade.has_value() && *filtro.idUnidade != eu.idUnidade())
104 return ResItens::falha(Erro::AcessoNegado,
105 "Consulta restrita a propria unidade");
106 idUnidade = eu.idUnidade();
107 } else if (perfil == Perfil::GestorRede || perfil == Perfil::Administrador) {
108 if (!filtro.idUnidade.has_value())
109 return ResItens::falha(Erro::UnidadeNaoInformada,
110 "Informe a unidade a consultar");
111 idUnidade = *filtro.idUnidade;
112 } else {
113 return ResItens::falha(Erro::AcessoNegado,
114 "Perfil sem permissao para consultar estoque");
115 }
116
117 Data hoje = Relogio::hoje();
118 const std::vector<Produto>& produtos = banco_->produtos();
119 const std::vector<LoteEstoque>& lotes = banco_->lotes();
120 std::vector<ItemEstoque> itens;
121
122 for (size_t i = 0; i < produtos.size(); ++i) {
123 const Produto& p = produtos[i];
124 if (!filtro.categoria.empty() && p.categoria() != filtro.categoria)
125 continue;
126
127 int valida = 0, vencida = 0;
128 bool temValidade = false;
129 Data proxima;
130
131 for (size_t j = 0; j < lotes.size(); ++j) {
132 const LoteEstoque& l = lotes[j];
133 if (l.idUnidade() != idUnidade || l.codigoProduto() != p.codigo()
134 || l.saldo() <= 0)
135 continue;
136
137 if (l.vencido(hoje)) {
138 vencida += l.saldo();
139 } else {
140 valida += l.saldo();
141 if (!temValidade || l.validade() < proxima) {
142 proxima = l.validade();
143 temValidade = true;
144 }
145 }
146 }
147
148 itens.push_back(ItemEstoque(p.codigo(), p.nome(), valida, vencida,
149 p.estoqueMinimo(), temValidade, proxima));
150 }
151
152 return ResItens::sucesso(itens);
153}
